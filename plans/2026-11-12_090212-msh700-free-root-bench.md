    return np.load(sorted(glob.glob(os.path.join(CLIPS, "*.npz")))[0])


def test_free_root_model_maps_every_clip_joint():
    m, z = br.build_model(XML), _clip()

    # The 2 ms sleep per step is a planted signal, not a tolerance.

### B3. Implement minimally: `bench_rollout.py`

```python
"""CPU mj_step cost of MS-Human-700 on the MuscleMimic free-root model, timed per 144 Hz frame.

Measures physics only: no policy, sensors, rendering, MJX or Warp. Random ctrl collapses the body,
so contact load is lower than in tracked motion.
Usage: MSH700_XML=... MSH700_CLIPS=... python bench_rollout.py [out.tsv]
"""
import gc
import glob
import importlib.util
import os
import pathlib
import sys
import time

import mujoco
import numpy as np

_here = pathlib.Path(__file__).resolve().parent
_root_py = _here.parents[1] / "3-interactor/musclemimic/musclemimic/environments/humanoids/ms_human_700_root.py"
_spec = importlib.util.spec_from_file_location("ms_human_700_root", _root_py)
root = importlib.util.module_from_spec(_spec)
_spec.loader.exec_module(root)

QPOS_WIDTH = {0: 7, 1: 4, 2: 1, 3: 1}  # mjtJoint: free, ball, slide, hinge
FRAME_HZ = 144.0                         # Steam Frame interactive target
PERIOD_MS = 1000.0 / FRAME_HZ            # 6.94 ms
WARMUP_FRAMES = 50                       # dropped: first-touch allocations and cache warm-up
CLIP_NAMES = ("crouch_walk", "run", "jump_forward", "crouch_down_up", "crawl")
REPEATS = 3
FRAMES = 1440                            # 10 s at 144 Hz


def build_model(xml):
    spec = mujoco.MjSpec.from_file(xml)
    root.prepare_spec(spec, {})          # sites do not enter dynamics
    return spec.compile()


def map_clip(model, names, types):
    """Return (clip_col, model_qposadr, width) per clip joint; raise if any joint is missing."""
    cols, cur, missing = [], 0, []
    for name, jtype in zip([str(n) for n in names], [int(t) for t in types]):
        jid = mujoco.mj_name2id(model, mujoco.mjtObj.mjOBJ_JOINT, name)
        if jid < 0 or model.jnt_type[jid] != jtype:
            missing.append(name)
        else:
            cols.append((cur, int(model.jnt_qposadr[jid]), QPOS_WIDTH[jtype]))
        cur += QPOS_WIDTH[jtype]
    if missing:
        raise ValueError(f"clip joints not in model: {missing}")
    return cols


def time_frames(model, clip, frames, seed=0, step=mujoco.mj_step):
    data = mujoco.MjData(model)
    for col, adr, width in map_clip(model, clip["joint_names"], clip["jnt_type"]):
        data.qpos[adr:adr + width] = clip["qpos"][0, col:col + width]
    mujoco.mj_forward(model, data)
    rng = np.random.default_rng(seed)
    lo, hi = model.actuator_ctrlrange[:, 0], model.actuator_ctrlrange[:, 1]
    frame_s = np.empty(frames)
    ncon = np.empty(frames, dtype=int)
    niter = np.empty(frames, dtype=int)
    nsteps = np.empty(frames, dtype=int)
    owed = 0.0
    gc.disable()
    try:
        for f in range(frames + WARMUP_FRAMES):
            data.ctrl[:] = rng.uniform(lo, hi)
            owed += 1.0 / FRAME_HZ
            t0 = time.perf_counter()
            n = 0
            while owed >= model.opt.timestep:
                step(model, data)
                owed -= model.opt.timestep
                n += 1
            if f >= WARMUP_FRAMES:
                k = f - WARMUP_FRAMES
                frame_s[k] = time.perf_counter() - t0
                ncon[k] = data.ncon
                niter[k] = data.solver_niter[0]
                nsteps[k] = n
    finally:
        gc.enable()
    if not np.isfinite(data.qpos).all():
        raise FloatingPointError("qpos went non-finite")
    return {"frame_s": frame_s, "ncon": ncon, "niter": niter, "nsteps": nsteps}


def summarize(frame_s, nsteps):
    # A 6.94 ms frame holds 3 or 4 steps of 2 ms. Mixing them inflates cs (about 0.14 by itself),
    # so judge only the worst-case frames, the ones holding the most steps.
    every = frame_s * 1e3
    ms = every[nsteps == nsteps.max()]
    cs = float(ms.std() / ms.mean())
    budget = PERIOD_MS / (1.0 + 4.0 * cs)
    return {"p50": float(np.percentile(ms, 50)), "p99": float(np.percentile(ms, 99)),
            "max": float(ms.max()), "cs": cs, "budget": budget,
            "realtime": (len(every) / FRAME_HZ) / (every.sum() / 1e3)}


def main(argv):
    xml, clips_dir = os.environ["MSH700_XML"], os.environ["MSH700_CLIPS"]
    out = argv[1] if len(argv) > 1 else "bench_rollout.tsv"
    model = build_model(xml)
    print(f"model nq={model.nq} nv={model.nv} nu={model.nu} dt={model.opt.timestep} frame_hz={FRAME_HZ}")
    rows = ["clip\trepeat\tp50_ms\tp99_ms\tmax_ms\tcs\tbudget_ms\trealtime\tslowest_frames"]
    for name in CLIP_NAMES:
        path = os.path.join(clips_dir, f"{name}.npz")
        clip = np.load(path)
        slow_sets = []
        for rep in range(REPEATS):
            r = time_frames(model, clip, FRAMES, seed=0)
            s = summarize(r["frame_s"], r["nsteps"])
            slow = np.argsort(r["frame_s"])[-5:][::-1]
            slow_sets.append(set(slow.tolist()))
            detail = ",".join(f"{i}:ncon{r['ncon'][i]}:it{r['niter'][i]}" for i in slow)
            print(f"{name} rep{rep}: p50 {s['p50']:.3f} p99 {s['p99']:.3f} max {s['max']:.3f} ms  "
                  f"cs {s['cs']:.3f} budget {s['budget']:.3f} ms  realtime {s['realtime']:.2f}x  slowest {detail}")
            rows.append(f"{name}\t{rep}\t{s['p50']:.4f}\t{s['p99']:.4f}\t{s['max']:.4f}\t{s['cs']:.4f}"
                        f"\t{s['budget']:.4f}\t{s['realtime']:.3f}\t{detail}")
        recur = set.intersection(*slow_sets)
        print(f"{name}: slowest frames recurring in all {REPEATS} repeats: {sorted(recur) or 'none (noise)'}")
    pathlib.Path(out).write_text("\n".join(rows) + "\n", encoding="utf-8")
    print(f"wrote {out}")


if __name__ == "__main__":
    main(sys.argv)
```

## Phase D: players wearing motion capture (replaces the earlier "Phase B2")

"Real time" means players in a virtual world wear motion capture, and each player's motion drives an MS-Human-700 body live. The body is not driven by random ctrl or by a clip-trained policy. The numbers that matter are:
- latency per player per mocap sample (p99, sample in to muscle state out),
- players per core at that latency.

Two workloads fit this. Phase B measures neither directly, and question 1 below decides which one to build:
- **Inverse (estimate muscles from the player's motion):** set qpos from the capture, take causal differences for qvel and qacc, run `mj_inverse`, then solve 700 bounded activations against the joint torques. There is no forward stepping, and the cost is paid once per mocap sample.
- **Forward (a simulated body tracks the player):** a controller or policy drives the 700 muscles at 500 Hz so the body follows the capture. The body can deviate from the player, for example when it is blocked by world contact.

### D2. Inverse-dynamics latency (TDD like B2 to B4)

Add `time_inverse` to `bench_rollout.py`. It uses backward differences only, so it is causal like a live stream:

```python
def time_inverse(model, clip, frames, hz=100.0):
    """Per-sample cost of mj_inverse from captured poses, plus the free-root residual wrench."""
    data = mujoco.MjData(model)
    q = np.tile(model.qpos0, (frames, 1))
    for col, adr, width in map_clip(model, clip["joint_names"], clip["jnt_type"]):
        q[:, adr:adr + width] = clip["qpos"][:frames, col:col + width]
    root_dof = int(model.jnt_dofadr[mujoco.mj_name2id(model, mujoco.mjtObj.mjOBJ_JOINT, "root")])
    dt = 1.0 / hz
    v_prev = np.empty(model.nv)
    cost = np.empty(frames - 2)
    residual = np.empty(frames - 2)
    for k in range(2, frames):
        t0 = time.perf_counter()
        data.qpos[:] = q[k]
        mujoco.mj_differentiatePos(model, data.qvel, dt, q[k - 1], q[k])
        mujoco.mj_differentiatePos(model, v_prev, dt, q[k - 2], q[k - 1])
        data.qacc[:] = (data.qvel - v_prev) / dt
        mujoco.mj_inverse(model, data)
        cost[k - 2] = time.perf_counter() - t0
        residual[k - 2] = np.linalg.norm(data.qfrc_inverse[root_dof:root_dof + 6])
    return {"frame_s": cost, "root_residual": residual}
```

The tests come first:
- `cost` has `frames - 2` entries, all finite.
- Planted negative: shift one clip frame 0.1 m in z, and the `root_residual` at that frame must exceed the 99th percentile of the unshifted run. Muscles cannot push the free root, so capture noise and impossible motion show up there. That residual is the quality measure for this path.

The clips were resampled offline with makima, which is smoother than a live stream. A live run needs a causal filter whose cutoff comes from measured tracker noise, not a picked constant. Record that as an open item. Do not add a filter in D2.
